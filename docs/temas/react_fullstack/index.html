CORS

Cross-Origin Resource Sharing

El problema


React (localhost:5173) ──fetch──▶ Express (localhost:3001)
                                        │
                                   ❌ BLOQUEADO
                                   "CORS policy error"

El navegador bloquea peticiones entre orígenes distintos por seguridad

La solución en Express

// Instalar: npm install cors
const cors = require('cors');
const express = require('express');
const app = express();

// Permitir peticiones desde React
app.use(cors({
  origin: 'http://localhost:5173'
}));

// O permitir cualquier origen (solo para desarrollo)
app.use(cors());

app.use(express.json());
// ... el resto de tus rutas

¿Cuándo ocurre CORS?

  • Diferente puerto: localhost:5173 → localhost:3001
  • Diferente dominio: mi-app.com → api.mi-app.com
  • Diferente protocolo: http → https

En producción se suele resolver con un proxy o mismo dominio

Configurar el backend (Express)

// backend/server.js
const express = require('express');
const cors = require('cors');
const { uid } = require('uid');

const app = express();
app.use(cors());
app.use(express.json());

let usuarios = [
  { id: uid(), name: 'Luis', level: 90 },
  { id: uid(), name: 'Ana', level: 100 }
];

app.get('/usuarios', (req, res) => {
  res.json(usuarios);
});

app.post('/usuario', (req, res) => {
  const nuevo = { ...req.body, id: uid() };
  usuarios.push(nuevo);
  res.status(201).json(nuevo);
});

app.delete('/usuario/:id', (req, res) => {
  usuarios = usuarios.filter(u => u.id !== req.params.id);
  res.json(usuarios);
});

app.listen(3001, () => console.log('API en http://localhost:3001'));

Conectando React con Express

GET: Obtener datos

// components/UserList.jsx
import { useState, useEffect } from 'react';

function UserList() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch('http://localhost:3001/usuarios')
      .then(res => res.json())
      .then(data => {
        setUsers(data);
        setLoading(false);
      });
  }, []);

  if (loading) return <p>Cargando usuarios...</p>;

  return (
    <ul>
      {users.map(user => (
        <li key={user.id}>
          {user.name} - Nivel {user.level}
        </li>
      ))}
    </ul>
  );
}

POST: Crear datos

// components/UserForm.jsx
import { useState } from 'react';

function UserForm({ onUserCreated }) {
  const [name, setName] = useState('');
  const [level, setLevel] = useState(1);

  const handleSubmit = async (e) => {
    e.preventDefault();
    const res = await fetch('http://localhost:3001/usuario', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name, level })
    });
    const newUser = await res.json();
    onUserCreated(newUser);
    setName('');
    setLevel(1);
  };

  return (
    <form onSubmit={handleSubmit}>
      <input value={name} onChange={e => setName(e.target.value)}
        placeholder="Nombre" required />
      <input type="number" value={level}
        onChange={e => setLevel(Number(e.target.value))} />
      <button type="submit">Crear Usuario</button>
    </form>
  );
}

DELETE: Eliminar datos

function UserItem({ user, onDelete }) {
  const handleDelete = async () => {
    await fetch(`http://localhost:3001/usuario/${user.id}`, {
      method: 'DELETE'
    });
    onDelete(user.id);
  };

  return (
    <li>
      <span>{user.name} - Nivel {user.level}</span>
      <button onClick={handleDelete}>Eliminar</button>
    </li>
  );
}

App.jsx: Uniendo todo

import { useState, useEffect } from 'react';
import UserForm from './components/UserForm';
import UserItem from './components/UserItem';

function App() {
  const [users, setUsers] = useState([]);

  useEffect(() => {
    fetch('http://localhost:3001/usuarios')
      .then(res => res.json())
      .then(setUsers);
  }, []);

  const handleCreated = (newUser) => {
    setUsers([...users, newUser]);
  };

  const handleDelete = (id) => {
    setUsers(users.filter(u => u.id !== id));
  };

  return (
    <div>
      <h1>Gestión de Usuarios</h1>
      <UserForm onUserCreated={handleCreated} />
      <ul>
        {users.map(user => (
          <UserItem key={user.id} user={user}
            onDelete={handleDelete} />
        ))}
      </ul>
    </div>
  );
}

Custom Hook para la API

Reutilizar la lógica de conexión

// hooks/useApi.js
import { useState, useCallback } from 'react';

const API_URL = 'http://localhost:3001';

function useApi() {
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  const request = useCallback(async (endpoint, options = {}) => {
    setLoading(true);
    setError(null);
    try {
      const res = await fetch(`${API_URL}${endpoint}`, {
        headers: { 'Content-Type': 'application/json' },
        ...options,
        body: options.body ? JSON.stringify(options.body) : undefined
      });
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      const data = await res.json();
      return data;
    } catch (err) {
      setError(err.message);
      return null;
    } finally {
      setLoading(false);
    }
  }, []);

  return { request, loading, error };
}

Proxy en Vite

Alternativa a CORS durante desarrollo

// vite.config.js
import { defineConfig } from 'vite';
import react from '@vitejs/plugin-react';

export default defineConfig({
  plugins: [react()],
  server: {
    proxy: {
      '/api': {
        target: 'http://localhost:3001',
        changeOrigin: true,
        rewrite: (path) => path.replace(/^\/api/, '')
      }
    }
  }
});
// Ahora en React puedes usar rutas relativas
fetch('/api/usuarios'); // Vite redirige a localhost:3001/usuarios

Patrones de Estado

Manejo profesional de datos del servidor

Los 3 estados de una petición

function UserList() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);  // 1. Cargando
  const [error, setError] = useState(null);      // 2. Error

  useEffect(() => {
    fetch('http://localhost:3001/usuarios')
      .then(res => {
        if (!res.ok) throw new Error('Error del servidor');
        return res.json();
      })
      .then(data => setUsers(data))             // 3. Éxito
      .catch(err => setError(err.message))
      .finally(() => setLoading(false));
  }, []);

  if (loading) return <p>Cargando...</p>;
  if (error) return <p>Error: {error}</p>;
  if (users.length === 0) return <p>No hay usuarios</p>;

  return <ul>{users.map(u => <li key={u.id}>{u.name}</li>)}</ul>;
}

Actualización optimista

const handleDelete = async (id) => {
  // 1. Actualizar UI inmediatamente (optimista)
  setUsers(users.filter(u => u.id !== id));

  try {
    // 2. Enviar al servidor
    const res = await fetch(`http://localhost:3001/usuario/${id}`, {
      method: 'DELETE'
    });
    if (!res.ok) throw new Error('Error al eliminar');
  } catch (err) {
    // 3. Si falla, revertir
    setUsers(users); // restaurar estado anterior
    alert('No se pudo eliminar');
  }
};

Errores Frecuentes

Checklist antes de pedir ayuda

  1. ¿El backend está corriendo? (npm start en terminal 1)
  2. ¿Instalaste cors en Express?
  3. ¿Usaste app.use(express.json())?
  4. ¿El fetch tiene Content-Type: application/json?
  5. ¿Inicializaste el estado con useState([]) y no useState()?
  6. ¿Tu useEffect tiene el array de dependencias []?
